cmake_minimum_required(VERSION 3.22)
project(NteSkinChangerLite VERSION 0.1.8 LANGUAGES CXX)
find_package(AnomalySDK CONFIG REQUIRED)
anomaly_add_plugin(skin_changer_lite SOURCES plugin.cpp cosmetics.cpp MANIFEST manifest.json PACKAGE_NAME SkinChanger)
target_compile_features(skin_changer_lite PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(skin_changer_lite PRIVATE /utf-8 /W4 /EHsc)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(cosmetics_tests cosmetics_tests.cpp)
  target_compile_features(cosmetics_tests PRIVATE cxx_std_20)
  add_test(NAME cosmetics_rules COMMAND cosmetics_tests)
endif()
